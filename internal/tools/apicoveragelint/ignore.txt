# Portainer API paths the provider leaves out on purpose, one per line with
# the reason after "#". apicoveragelint fails on any spec path that is neither
# called by the provider nor listed here, so a new Portainer release cannot add
# an API the provider silently ignores. Remove a line once the path is covered.

# Add-ons' own machine APIs: the add-ons call them themselves.
/addon-data/v1                      # key-value store of an add-on, written by the add-on
/addon-data/v1/{key}                # key-value store of an add-on, written by the add-on
/addon-user-data/v1                 # per-user store of an add-on, written by the add-on
/addon-user-data/v1/{key}           # per-user store of an add-on, written by the add-on
/addon-store/v1/capabilities        # add-on runtime API
/addon-store/v1/config              # add-on runtime API; configuration goes through portainer_addon_config
/addon-store/v1/config/{key}        # add-on runtime API; configuration goes through portainer_addon_config
/addons/{id}/config/{key}           # single-key patch; portainer_addon_config replaces the whole config

# Sessions, interactive and streaming endpoints.
/auth/logout                        # session handling
/auth/oauth/validate                # OAuth login flow of the UI
/websocket/attach                   # interactive
/websocket/exec                     # interactive
/websocket/kubernetes-shell         # interactive
/websocket/node-shell               # interactive
/websocket/pod                      # interactive

/endpoints/{id}/edge/charts         # Edge agent report
/endpoints/{id}/edge/charts/statuses # Edge agent report
/endpoints/{id}/edge/gitops/statuses # Edge agent report
/endpoints/{id}/edge/gitops/{deploymentId} # Edge agent fetch of a GitOps deployment
/endpoints/{id}/edge/jobs/{jobID}/logs # Edge agent upload of job logs
/endpoints/{id}/edge/policies/statuses # Edge agent report
/endpoints/{id}/edge/stacks/{stackId} # Edge agent fetch of a stack
/edge_configurations/{id}/{state}   # Edge agent report
/edge_stacks/{id}/status            # Edge agent report
/edge_stacks/{id}/logs/{endpoint_id} # Edge agent log collection
/edge_stacks/{id}/logs/{endpoint_id}/file # Edge agent log download

/kubernetes/{id}/namespaces/{namespace}/applications/{kind}/{name}/restart # runtime action
/kubernetes/{id}/namespaces/{namespace}/pods/{name}/restart # runtime action
/kubernetes/{id}/namespaces/{namespace}/pods/{name} # runtime object; pods belong to their workload
/omni/{credentialID}/cluster/provision/cancel # cancels a running provisioning
/support/download                   # support bundle download
/useractivity/authlogs.csv          # CSV export; portainer_user_activity reads the JSON
/useractivity/logs.csv              # CSV export; portainer_user_activity reads the JSON
/system/update                      # updates the Portainer server itself
/system/upgrade                     # upgrades the Portainer server itself
/status                             # deprecated alias of /system/status
/registries/{id}/repositories/{repositoryName}/tags # registry browsing
/registries/{id}/repositories/{repositoryName}/tags/details # registry browsing
/registries/{id}/ecr/repositories/{repositoryName} # ECR browsing and cleanup
/registries/{id}/ecr/repositories/{repositoryName}/tags # ECR browsing and cleanup
